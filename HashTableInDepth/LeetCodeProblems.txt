LC 242. Valid Anagram:
Solution:
class Solution {
    public boolean isAnagram(String s, String t) {
        // 1. Anagrams must have the same length.
        if (s.length() != t.length()) {
            return false;
        }

        // 2. Create a frequency map for characters 'a'-'z'.
        int[] charCounts = new int[26];

        // 3. Count character frequencies for the first string.
        for (char c : s.toCharArray()) {
            charCounts[c - 'a']++;
        }

        // 4. Decrement frequencies based on the second string.
        for (char c : t.toCharArray()) {
            // Decrement the count for the current character.
            charCounts[c - 'a']--;

            // If a count becomes negative, it means t has more of this
            // character than s, so they can't be anagrams.
            if (charCounts[c - 'a'] < 0) {
                return false;
            }
        }

        // 5. If the loop completes, the strings are anagrams.
        return true;
    }
}

LC 383: Ransom Note

class Solution {
    public boolean canConstruct(String ransomNote, String magazine) {
        // Optimization: If the note requires more letters than the magazine has, it's impossible.
        if (ransomNote.length() > magazine.length()) {
            return false;
        }

        // Create an array of 26 zeros to represent the letters 'a' through 'z'
        int[] charCounts = new int[26];

        // Step 1: Build the inventory from the magazine
        for (char c : magazine.toCharArray()) {
            // c - 'a' maps 'a' to 0, 'b' to 1, ..., 'z' to 25
            charCounts[c - 'a']++;
        }

        // Step 2 & 3: Read the note and consume letters
        for (char c : ransomNote.toCharArray()) {
            // If we have 0 of this letter left, we can't build the note
            if (charCounts[c - 'a'] == 0) {
                return false;
            }

            // Consume the letter
            charCounts[c - 'a']--;
        }

        // Step 4: If we make it through the whole note, we had enough letters
        return true;
    }
}

LC 387. First Unique Character in a String

class Solution {
    public int firstUniqChar(String s) {
        // Step 1: Create the frequency array
        int[] charCounts = new int[26];

        // Step 2: First pass - count occurrences of each character
        for (char c : s.toCharArray()) {
            charCounts[c - 'a']++;
        }

        // Step 3: Second pass - find the first character with a count of 1
        for (int i = 0; i < s.length(); i++) {
            if (charCounts[s.charAt(i) - 'a'] == 1) {
                return i; // Return the index immediately
            }
        }

        // Step 4: If no unique character was found, return -1
        return -1;
    }
}

LC 217. Contains Duplicate
class Solution {
    public boolean containsDuplicate(int[] nums) {
        // {0,1,2,3,3,4,5,5,6,7,8,8}
        Arrays.sort(nums);

        for(int i=0;i<nums.length-1;i++) {
            if(nums[i] == nums[i+1]) {
                return true;
            }
        }
        return false;
    }
}
